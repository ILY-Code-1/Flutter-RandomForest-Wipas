' == Activity: UC01 Login ==
@startuml act_UC01
|Pengguna|
start
:Buka aplikasi (Splash);
:Isi email dan password;
|Aplikasi WIPAS|
:Periksa akun dan role;
if (Akun valid dan punya role?) then (ya)
  :Catat awal sesi (24 jam);
  :Tampilkan Beranda sesuai role;
else (tidak)
  :Tampilkan "Email atau password salah";
endif
stop
@enduml
