' == Sequence: UC01 Login ==
@startuml seq_UC01
autoactivate off
actor Pengguna
boundary LoginScreen
control AuthController
entity users
Pengguna -> LoginScreen : isi email & password, tekan Masuk
activate LoginScreen
LoginScreen -> AuthController : signIn(email, password)
activate AuthController
AuthController -> users : get(users/{uid})
activate users
users --> AuthController : role, display_name, wh_id
deactivate users
alt akun valid dan role dikenal
  AuthController --> LoginScreen : AuthState signedIn
  LoginScreen --> Pengguna : buka /beranda (roleGuard)
else kredensial salah
  AuthController --> LoginScreen : AuthFailure
  LoginScreen --> Pengguna : "Email atau password salah"
end
deactivate AuthController
deactivate LoginScreen
@enduml

' == Sequence: UC02 Reset Kata Sandi ==
@startuml seq_UC02
actor Pengguna
boundary ForgotPasswordScreen
control AuthController
entity "Firebase Auth" as FirebaseAuth
Pengguna -> ForgotPasswordScreen : isi email, tekan Kirim tautan reset
activate ForgotPasswordScreen
alt format email valid
  ForgotPasswordScreen -> AuthController : sendPasswordReset(email)
  activate AuthController
  AuthController -> FirebaseAuth : sendPasswordResetEmail(email)
  activate FirebaseAuth
  FirebaseAuth --> AuthController : terkirim
  deactivate FirebaseAuth
  AuthController --> ForgotPasswordScreen : selesai
  deactivate AuthController
  ForgotPasswordScreen --> Pengguna : "Tautan reset telah dikirim ke email"
else email tidak valid
  ForgotPasswordScreen --> Pengguna : "Masukkan email yang valid"
end
deactivate ForgotPasswordScreen
@enduml

' == Sequence: UC15 Menjalankan Batch Prediksi ==
@startuml seq_UC15

@enduml
